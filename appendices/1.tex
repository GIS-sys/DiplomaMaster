\subsection{Код для поиска статистики по операциям в датасете}

\begin{lstlisting}
import json
import os
from collections import defaultdict, Counter

DATASET_ROOT = "~/data/"
TYPE = 1

def get_operations_from_json(json_path):
    """Extracts a list of operation types from a JSON file of file history"""
    with open(json_path, 'r', encoding='utf-8') as f:
        data = json.load(f)
    operations = data.get("entities", [])
    return [op.get("feature") for op in operations.values() if "feature" in op]

def get_operations_from_steps(steps_path):
    """Extracts a list of operation types from STEPS file of file history"""
    with open(steps_path, 'r', encoding='utf-8') as f:
        data = f.read()
    operations = data.split("Feature = ")
    return [op[:(op.index("\n") if "\n" in op else len(op))] for op in operations if op]


def compute_statistics():
    model_counts = Counter()
    total_models = 0
    op_occurrences = defaultdict(list)
    
    for root, dirs, files in os.walk(DATASET_ROOT + ("old/" if TYPE == 0 else "new/")):
        if TYPE == 0:
            # Find JSON file in each folder
            json_files = [f for f in files if f.endswith(".json")]
            if not json_files:
                continue
            json_path = os.path.join(root, json_files[0])
            try:
                ops = get_operations_from_json(json_path)
            except Exception as e:
                print(f"Error in {json_path}: {e}")
                continue
        else:
            steps_files = [f for f in files if f.endswith("Steps.txt")]
            if not steps_files:
                continue
            steps_path = os.path.join(root, steps_files[0])
            try:
                ops = get_operations_from_steps(steps_path)
            except Exception as e:
                print(f"Error in {steps_path}: {e}")
                continue
        
        total_models += 1
        # For each operation type count ocurrences
        for op_type in set(ops):
            count = ops.count(op_type)
            if count > 0:
                model_counts[op_type] += 1
                op_occurrences[op_type].append(count)
    
    print(f"Total models after filtration: {total_models}\n")
    print("Operation, Models, Fraction, Operations per model")
    for op in op_occurrences.keys():
        cnt_models = model_counts[op]
        fraction = cnt_models / total_models * 100
        avg_per_model = sum(op_occurrences[op]) / len(op_occurrences[op]) if op_occurrences[op] else 0
        print(f"{op}, {cnt_models}, {fraction}, {avg_per_model}")

if __name__ == "__main__":
    compute_statistics()
\end{lstlisting}



\subsection{Код для поиска статистики по очищенной выборке}

\begin{lstlisting}
import os
from pathlib import Path

root = Path("~/data/")
min_stl = 2

all_folders = []
errors_multiplejson = []   # folders with >1 JSON
errors_nojson = []         # folders without json
errors_diffstlstp = []     # amount of STL != amount of STEP
errors_nostl = []          # no STL files
errors_stlfew = []         # less STL files than min_stl
valid_folders = []         # folders which passed all checks

for current_dir, subdirs, files in os.walk(root):
    if subdirs:
        continue

    json_files = [f for f in files if f.endswith('.json')]
    stl_files  = [f for f in files if f.endswith('.stl')]
    stp_files  = [f for f in files if f.endswith('.stp')]

    current_dir = str(current_dir)
    all_folders.append(current_dir)
    if len(json_files) > 1:
        errors_multiplejson.append(current_dir)
    elif len(json_files) == 0:
        errors_nojson.append(current_dir)
    elif len(stl_files) != len(stp_files):
        errors_diffstlstp.append(current_dir)
    elif len(stl_files) == 0:
        errors_nostl.append(current_dir)
    elif len(stl_files) < min_stl:
        errors_stlfew.append(current_dir)
    else:
        valid_folders.append(current_dir)

print("=== DATASET STATISTICS ===")
print(f"Total folders:                      {len(all_folders)}")
print(f"  several json:                     {len(errors_multiplejson)}")
print(f"  no json:                          {len(errors_nojson)}")
print(f"  different amount of STL and STEP: {len(errors_diffstlstp)}")
print(f"  no STL:                           {len(errors_nostl)}")
print(f"  STL amount less than {min_stl}:   {len(errors_stlfew)}")
print(f"Correct:                            {len(valid_folders)}")
\end{lstlisting}

\subsection{Код для замера скорости работы на Linux}

\begin{lstlisting}
tasks=("classification" "classification" "segmentation" "segmentation" "segmentation")
operations=("dip_a_cls_chamfer" "dip_b_cls_chamfer" "dip_c_seg_chamfer" "dip_d_seg_chamfer" "dip_e_seg_chamfer")
runs=10

for ((i=0; i<${#tasks[@]} && i<${#operations[@]}; i++)); do
    task="${tasks[$i]}"
    operation="${operations[$i]}"

    for use_cache in false true; do
        [[ $use_cache == true ]] && cache_arg="--allow_cache" || cache_arg=""

        for use_subdiv in false true; do
            [[ $use_subdiv == true ]] && subdiv_arg="--allow_cache" || subdiv_arg=""
            cfg="task=$task, op=$operation, cache=$use_cache, subdiv=$use_subdiv"
            printf "\nRunning: %s\n" "$cfg"

            times=()
            for ((run=1; run<=runs; run++)); do
                start=$(date +%s%N)
                python main.py \
                    --path_to_input ./MY/test.stp \
                    --target_folder /tmp/output \
                    --task "$task" \
                    --operation "$operation" \
                    $subdiv_arg -c \
                    --remove_largest_area 0.0 \
                    $cache_arg \
                    > /dev/null 2>&1
                end=$(date +%s%N)
                ms=$(echo "scale=3; ($end - $start) / 1000000" | bc)
                times+=("$ms")
                echo "$ms"
            done

            avg=$(printf "%s\n" "${times[@]}" | awk '{sum+=$1} END {printf "%.3f", sum/NR}')
            printf "Average: %s ms\n" "$avg"
        done
    done
done
\end{lstlisting}

\subsection{Код для замера скорости работы на Windows}

\begin{lstlisting}
$tasks = @('classification', 'classification', 'segmentation', 'segmentation', 'segmentation')
$operations = @('a', 'b', 'c', 'd', 'e')
$cacheFlags = @($false, $true)
$subdivFlags = @($false, $true)

for ($i = 0; $i -lt $tasks.Count -and $i -lt $operations.Count; $i++) {
    $task = $tasks[$i]
    $operation = $operations[$i]
    foreach ($useCache in $cacheFlags) {
      foreach ($useSubdiv in $subdivFlags) {
        $cacheArg = if ($useCache) { '--allow_cache' } else { '' }
        $subdivArg = if ($useSubdiv) { '-d' } else { '' }
        $configName = "task=$task, op=$operation, cache=$useCache, useSubdiv=$useSubdiv"
        Write-Host "`nRunning: $configName" -ForegroundColor Yellow
        $times = 1..10 | ForEach-Object {
            (Measure-Command {
                .\main.exe --path_to_input .\_internal\sample\test.stp --target_folder output --task $task --operation $operation $subdivArg -c --remove_largest_area 0.0 $cacheArg
            }).TotalMilliseconds
        }
        $times | ForEach-Object { Write-Host $_ }
        $avg = ($times | Measure-Object -Average).Average
        Write-Host "Average: $avg ms" -ForegroundColor Green
      }
    }
}
\end{lstlisting}
