\section{Для обучения нейронных сетей}

\subsection{Минимальная конфигурация}
\begin{itemize}
    \item GPU: NVIDIA RTX 3060 (12 GB VRAM) или аналогичная с поддержкой CUDA.
    \item RAM: 32 GB.
    \item SSD: 100 GB свободного места.
    \item Процессор: 8 ядер, 3.0 GHz.
\end{itemize}

\subsection{Оптимальная конфигурация}
\begin{itemize}
    \item GPU: NVIDIA RTX 4090 (24 GB VRAM) или A6000.
    \item RAM: 64 GB и более.
    \item NVMe SSD: от 500 GB.
    \item Процессор: 16 ядер, 3.5 GHz.
\end{itemize}

\section{Для инференса}
\begin{itemize}
    \item GPU: не требуется.
    \item RAM: 8 GB.
    \item SSD: 4 GB свободного места.
    \item Процессор: 4 ядра, 2.4 GHz.
\end{itemize}
